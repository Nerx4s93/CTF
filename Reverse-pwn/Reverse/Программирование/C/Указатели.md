## Арифметика указателей
Прибавляется размер типа данных.
``` C
int *a = &b;
a++; // прибавит 4 (размер типа int = байта)
```

``` C
#include <stdio.h>
 
int main(void)
{
    int a = 10;
    int b = 23;
    int *pa = &a;
    int *pb = &b;
    ptrdiff_t c = pa - pb;  // разница между адресами
     
    printf("pa=%p \n", (void*)pa);
    printf("pb=%p \n", (void*)pb);
    printf("c=%lld \n", c);
     
    return 0;
}
```

## Константы
``` C
#include <stdio.h>
 
 
int main(void)
{
    const int cx = 10;
    
    *(int*) &cx = 20;   
    printf("cx: %d\n", cx);  // cx: 20
 
    // то же самое, только разбито на два этапа
    int* cx_p = &cx;    
    *cx_p = 66;
    printf("cx: %d\n", cx); // cx: 66
    return 0;
}
```

``` C
#include <stdio.h>
 
int main(void)
{
    const int a = 10;
    const int *pa = &a;
    printf("address=%p \t value=%d \n", pa, *pa);
     
    return 0;
}
```
